
import { createHmac, timingSafeEqual } from 'node:crypto';

const COOKIE = 'mq_admin';

function sign(value) {
  return createHmac('sha256', process.env.ADMIN_API_SECRET)
    .update(value)
    .digest('hex');
}

function validToken(token) {
  if (!token || !process.env.ADMIN_API_SECRET) return false;

  const parts = token.split('.');
  if (parts.length !== 2) return false;

  const [expires, signature] = parts;
  if (!/^\d+$/.test(expires) || Number(expires) < Date.now()) {
    return false;
  }

  const expected = sign(expires);
  const a = Buffer.from(signature);
  const b = Buffer.from(expected);

  return a.length === b.length && timingSafeEqual(a, b);
}

function getToken(req) {
  const cookies = (req.headers.cookie || '').split(';');
  const item = cookies.find(c => c.trim().startsWith(COOKIE + '='));
  return item ? decodeURIComponent(item.trim().slice(COOKIE.length + 1)) : '';
}

export default async function handler(req, res) {
  res.setHeader('Cache-Control', 'no-store');

  if (req.method === 'GET') {
    return res.status(200).json({ authenticated: validToken(getToken(req)) });
  }

  if (req.method !== 'POST') {
    res.setHeader('Allow', 'GET, POST');
    return res.status(405).json({ error: 'Method not allowed' });
  }

  if (req.body?.action === 'logout') {
    res.setHeader(
      'Set-Cookie',
      `${COOKIE}=; HttpOnly; Secure; SameSite=Strict; Path=/; Max-Age=0`
    );
    return res.status(200).json({ ok: true });
  }

  const password = req.body?.password;
  if (
    !process.env.ADMIN_PASSWORD ||
    !process.env.ADMIN_API_SECRET
  ) {
    return res.status(500).json({ error: 'Admin sozlamalari kiritilmagan' });
  }

  if (
    typeof password !== 'string' ||
    password.length > 200 ||
    password !== process.env.ADMIN_PASSWORD
  ) {
    return res.status(401).json({ error: 'Parol noto‘g‘ri' });
  }

  const expires = String(Date.now() + 7 * 24 * 60 * 60 * 1000);
  const token = `${expires}.${sign(expires)}`;

  res.setHeader(
    'Set-Cookie',
    `${COOKIE}=${encodeURIComponent(token)}; HttpOnly; Secure; SameSite=Strict; Path=/; Max-Age=604800`
  );

  return res.status(200).json({ ok: true });
}
